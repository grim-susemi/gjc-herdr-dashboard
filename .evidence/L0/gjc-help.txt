gjc v0.18.11

USAGE
  $ gjc [COMMAND]

ARGUMENTS
  MESSAGES   Messages to send (prefix files with @)

FLAGS
      --model=<value>                 Model to use (fuzzy match: "opus", "gpt-5.2", or "openai/gpt-5.2")
      --smol=<value>                  Smol/fast model for lightweight tasks (or GJC_SMOL_MODEL env)
      --slow=<value>                  Slow/reasoning model for thorough analysis (or GJC_SLOW_MODEL env)
      --plan=<value>                  Plan model for architectural planning (or GJC_PLAN_MODEL env)
      --mpreset=<value>               Model profile preset to activate for this session
      --default                       Persist --mpreset as the default model profile
      --provider=<value>              Provider to use (legacy; prefer --model)
      --api-key=<value>               API key (defaults to env vars)
      --credential=<value>            Stored credential selector: email:<addr>, id:<n>, account:<id>, project:<id>, or provider/email:<addr>
      --prefer-credential=<value>     Preferred stored credential with quota fallback: email:<addr>, id:<n>, account:<id>, project:<id>, or provider/id:<n>
      --system-prompt=<value>         System prompt (default: coding assistant prompt)
      --append-system-prompt=<value>  Append text or file contents to the system prompt
      --mcp-config=<value>            Tools-only MCP config file (absolute path)
      --no-mcp                        Disable conventional MCP autoload (native user ~/.gjc/agent/mcp.json and project .gjc/mcp.json registrations)
      --clipboard-transport=<value>   Clipboard transport: auto (default), native, osc52, or ssh
      --clipboard-ssh-host=<value>    SSH host alias for --clipboard-transport ssh (from ~/.ssh/config)
      --allow-home                    Allow starting in ~ without auto-switching to a temp dir
      --mode=<value>                  Output mode: text (default), json, or acp
  -p, --print                         Non-interactive mode: process prompt and exit
  -c, --continue                      Continue previous session
  -r, --resume[=<value>]              Resume a session (by ID prefix, path, or picker if omitted)
      --fork=<value>                  Fork a session (by ID prefix or path)
  -w, --worktree[=<value>]            Launch in a managed worktree (optional branch name)
      --session-dir=<value>           Explicit session storage directory and lookup override (default uses managed v2 workspace scope)
      --no-session                    Don't save session (ephemeral)
      --models=<value>                Comma-separated model patterns for Alt+N cycling
      --no-tools                      Disable all built-in tools
      --no-lsp                        Disable LSP tools, formatting, and diagnostics
      --no-pty                        Disable PTY-based interactive bash execution
      --tmux                          Launch interactive startup inside tmux
      --tools=<value>                 Comma-separated list of tools to enable (default: all)
      --thinking=<value>              Set thinking level: off, minimal, low, medium, high, xhigh, max
      --no-rules                      Disable rules discovery and loading
      --export=<value>                Export session file to HTML and exit
      --list-models[=<value>]         List available models (with optional fuzzy search)
      --no-title                      Disable title auto-generation
      --master                        Launch an interactive master session
      --scope=<value>                 Master scope: repo (default), pwd, or global

EXAMPLES
  # Interactive mode
    gjc
  # Interactive mode with initial prompt
    gjc "List all .ts files in src/"
  # Include files in initial message
    gjc @prompt.md @image.png "What color is the sky?"
  # Non-interactive mode (process and exit)
    gjc -p "List all .ts files in src/"
  # Continue previous session
    gjc --continue "What did we discuss?"
  # Launch in a sibling git worktree
    gjc --worktree
  # Use different model (fuzzy matching)
    gjc --model opus "Help me refactor this code"
  # Limit model cycling to specific models
    gjc --models claude-sonnet,claude-haiku,gpt-4o
  # Pin a stored credential for this session
    gjc --credential email:me@example.com
  # Prefer a stored credential, falling back on quota limits
    gjc --prefer-credential id:15
  # Activate a model profile for this session
    gjc --mpreset codex-medium
  # Persist a model profile as the default
    gjc --mpreset opencodego --default
  # Export a session file to HTML
    gjc --export ~/.gjc/agent/sessions/--path--/session.jsonl

Commands:
  gjc [prompt]             - Start an interactive coding session (default launch command)
  gjc launch               - Start an explicit launch/session workflow
  gjc setup                - Install GJC defaults or optional dependencies
  gjc session              - List, inspect, create, remove, or attach sessions
  gjc state                - Inspect or manage persisted GJC state
  gjc auth-broker         - Manage the auth-broker (credential vault)
  gjc accounts             - List, check, pin, and remove stored OAuth accounts
  gjc models presets       - Inspect, refresh, pin, rollback, or disable the signed model preset registry
  gjc auth-gateway        - Run an auth-gateway forward proxy
  gjc harness              - Run harness control-plane commands
  gjc coordinator          - Manage coordinator/runtime coordination helpers
  gjc ultragoal            - Run durable goal execution workflow
  gjc ralplan              - Run consensus planning workflow
  gjc autoresearch         - Run autonomous research workflow
  gjc deep-interview       - Run requirements interview workflow
  gjc skills               - List/read embedded workflow skills
  gjc config               - List, get, and set configuration values
  gjc stats                - View AI usage statistics (tokens, cost, requests)
  gjc notify               - Send or test notifications
  gjc daemon               - Manage background daemon helpers
  gjc mcp                  - Manage MCP server registrations
  gjc mcp-serve            - Serve the MCP integration endpoint
  gjc contribute-pr        - Prepare contribution/PR workflow artifacts
  gjc migrate              - Run migration helpers
  gjc update               - Update GJC installation artifacts
  gjc plugin               - Install, remove, and list plugins
  gjc web-search           - Search the web from the CLI (alias: q)
  gjc codex-native-hook    - Run Codex native hook integration
  gjc gc                   - Run garbage-collection/cleanup helpers
  gjc quick-lane         - Classify a task into quick lane or deep path
  gjc <command> --help     - Show command-specific help

Environment Variables:
  # Core Providers
  ANTHROPIC_API_KEY          - Anthropic Claude models
  ANTHROPIC_OAUTH_TOKEN      - Anthropic OAuth (takes precedence over API key)
  CLAUDE_CODE_USE_FOUNDRY    - Enable Anthropic Foundry mode (uses Foundry endpoint + mTLS)
  FOUNDRY_BASE_URL           - Anthropic Foundry base URL (e.g., https://<foundry-host>)
  ANTHROPIC_FOUNDRY_API_KEY  - Anthropic token used as Authorization: Bearer <token> in Foundry mode
  ANTHROPIC_CUSTOM_HEADERS   - Extra Foundry headers (e.g., "user-id: USERNAME")
  CLAUDE_CODE_CLIENT_CERT    - Client certificate (PEM path or inline PEM) for mTLS
  CLAUDE_CODE_CLIENT_KEY     - Client private key (PEM path or inline PEM) for mTLS
  NODE_EXTRA_CA_CERTS        - CA bundle path (or inline PEM) for server certificate validation
  OPENAI_API_KEY             - OpenAI GPT models
  GEMINI_API_KEY             - Google Gemini models
  GITHUB_TOKEN               - GitHub Copilot (or GH_TOKEN, COPILOT_GITHUB_TOKEN)

  # Additional LLM Providers
  AZURE_OPENAI_API_KEY       - Azure OpenAI models
  GROQ_API_KEY               - Groq models
  CEREBRAS_API_KEY           - Cerebras models
  XAI_API_KEY                - xAI Grok models
  OPENROUTER_API_KEY         - OpenRouter aggregated models
  KILO_API_KEY               - Kilo Gateway models
  MISTRAL_API_KEY            - Mistral models
  ZAI_API_KEY                - z.ai models (ZhipuAI/GLM)
  JUNIE_API_KEY              - JetBrains AI models (Junie; Claude via JetBrains AI)
  MINIMAX_API_KEY            - MiniMax models
  OPENCODE_API_KEY           - OpenCode Zen/OpenCode Go models
  CURSOR_ACCESS_TOKEN        - Cursor AI models
  AI_GATEWAY_API_KEY         - Vercel AI Gateway

  # Cloud Providers
  AWS_PROFILE                - AWS Bedrock (or AWS_ACCESS_KEY_ID + AWS_SECRET_ACCESS_KEY)
  GOOGLE_CLOUD_PROJECT       - Google Vertex AI (requires GOOGLE_CLOUD_LOCATION)
  GOOGLE_APPLICATION_CREDENTIALS - Service account for Vertex AI

  # Search & Tools
  EXA_API_KEY                - Exa web search
  BRAVE_API_KEY              - Brave web search
  PERPLEXITY_API_KEY         - Perplexity web search (API)
  PERPLEXITY_COOKIES         - Perplexity web search (session cookie)
  TAVILY_API_KEY             - Tavily web search
  ANTHROPIC_SEARCH_API_KEY   - Anthropic search provider

  # Configuration
  GJC_CODING_AGENT_DIR       - Session storage directory (default: ~/.gjc/agent)
  GJC_UI_LANGUAGE            - Explicit first-use onboarding language (en, ko, zh, or ja); invalid values use English without prompting
  GJC_PACKAGE_DIR            - Override package directory (for Nix/Guix store paths)
  GJC_SMOL_MODEL              - Override smol/fast model (see --smol)
  GJC_SLOW_MODEL              - Override slow/reasoning model (see --slow)
  GJC_PLAN_MODEL              - Override planning model (see --plan)
  GJC_MODEL_PRESET_REGISTRY_URL - HTTPS latest.json override for the signed preset registry
  GJC_MODEL_PRESET_REGISTRY_DISABLED - Disable registry fetch/use (1/true/yes/on)
  GJC_NO_PTY                  - Disable PTY-based interactive bash execution
  --tmux                       - Launch interactive startup inside a fresh tmux session
  gjc session                  - List, inspect, create, remove, or attach tagged GJC-managed tmux sessions
  GJC_LAUNCH_POLICY           - Launch policy for --tmux startup: tmux or direct
  GJC_TMUX_SESSION            - Explicit tmux session name override for --tmux startup
  GJC_TMUX_PROFILE            - Apply GJC tmux scroll/mouse/clipboard profile to --tmux sessions (set 0/off to skip)
  GJC_MOUSE                   - Apply tmux copy-mode mouse capture in --tmux sessions (set 0/off to skip)

  For complete environment variable reference, see:
  docs/environment-variables.md
